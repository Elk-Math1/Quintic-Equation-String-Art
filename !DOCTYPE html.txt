<body>
    <div class="container">
        <div class="info">
            <h1>⚽ 탑뷰 축구 게임 ⚽</h1>
            <div class="score-board">
                <div class="player1-score">플레이어 1: <span id="scoreLeft">0</span></div>
                <div class="player2-score">플레이어 2: <span id="scoreRight">0</span></div>
            </div>
            <div class="controls">
                <div class="control-p1">플레이어 1: W/A/S/D 이동, C 스프린트</div>
                <div class="control-p2">플레이어 2: 화살표 키 이동, Shift 스프린트</div>
                <div id="startHint" class="start-hint">W 또는 ↑ 키를 눌러 시작!</div>
                <div class="tip">💡 스프린트로 빠르게 이동하고 강하게 차세요! (스태미나 소모)</div>
            </div>
        </div>
        <div class="canvas-wrapper">
            <canvas id="gameCanvas" width="1056" height="792"></canvas>
        </div>
        <button onclick="resetGame()">게임 초기화</button>
    </div>

    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        
        // 기본 크기
        const SCALE = 0.66;
        const CANVAS_WIDTH = 1600 * SCALE;
        const CANVAS_HEIGHT = 1200 * SCALE;

        let score = { left: 0, right: 0 };
        let gameStarted = false;
        let keys = {};

        const game = {
            player1: { x: 400 * SCALE, y: 600 * SCALE, stamina: 100, sprinting: false },
            player2: { x: 1200 * SCALE, y: 600 * SCALE, stamina: 100, sprinting: false },
            ball: { x: 800 * SCALE, y: 600 * SCALE, vx: 0, vy: 0 },
            moveSpeed: 3.5 * SCALE,
            sprintSpeed: 6 * SCALE,
            ballRadius: 12 * SCALE,
            playerRadius: 20 * SCALE,
            fieldWidth: 1600 * SCALE,
            fieldHeight: 1200 * SCALE,
            goalWidth: 180 * SCALE,
            goalDepth: 40 * SCALE,
            maxStamina: 100,
            staminaDrainRate: 0.3,
            staminaKickCost: 25,
            staminaRegenRate: 0.15
        };

        window.addEventListener('keydown', (e) => {
            keys[e.key] = true;
            if (!gameStarted && (e.key === 'w' || e.key === 'W' || e.key === 'ArrowUp')) {
                gameStarted = true;
                document.getElementById('startHint').style.display = 'none';
            }
        });

        window.addEventListener('keyup', (e) => {
            keys[e.key] = false;
        });

        function checkCircleCollision(c1, c2, r1, r2) {
            const dx = c1.x - c2.x;
            const dy = c1.y - c2.y;
            const dist = Math.sqrt(dx * dx + dy * dy);
            return dist < r1 + r2;
        }

        function resetBall() {
            game.ball.x = game.fieldWidth / 2;
            game.ball.y = game.fieldHeight / 2;
            game.ball.vx = 0;
            game.ball.vy = 0;
        }

        function resetPlayers() {
            game.player1.x = 400 * SCALE;
            game.player1.y = 600 * SCALE;
            game.player1.stamina = 100;
            game.player1.sprinting = false;
            game.player2.x = 1200 * SCALE;
            game.player2.y = 600 * SCALE;
            game.player2.stamina = 100;
            game.player2.sprinting = false;
        }

        function gameLoop() {
            if (!gameStarted) return;

            // Sprint 처리
            game.player1.sprinting = (keys['c'] || keys['C']) && game.player1.stamina > 0;
            game.player2.sprinting = keys['Shift'] && game.player2.stamina > 0;

            const p1Speed = game.player1.sprinting ? game.sprintSpeed : game.moveSpeed;
            const p2Speed = game.player2.sprinting ? game.sprintSpeed : game.moveSpeed;

            // 플레이어 1 이동
            if (keys['a'] && game.player1.x > game.playerRadius) game.player1.x -= p1Speed;
            if (keys['d'] && game.player1.x < game.fieldWidth - game.playerRadius) game.player1.x += p1Speed;
            if (keys['w'] && game.player1.y > game.playerRadius) game.player1.y -= p1Speed;
            if (keys['s'] && game.player1.y < game.fieldHeight - game.playerRadius) game.player1.y += p1Speed;

            // 플레이어 2 이동
            if (keys['ArrowLeft'] && game.player2.x > game.playerRadius) game.player2.x -= p2Speed;
            if (keys['ArrowRight'] && game.player2.x < game.fieldWidth - game.playerRadius) game.player2.x += p2Speed;
            if (keys['ArrowUp'] && game.player2.y > game.playerRadius) game.player2.y -= p2Speed;
            if (keys['ArrowDown'] && game.player2.y < game.fieldHeight - game.playerRadius) game.player2.y += p2Speed;

            // 스태미나 관리
            game.player1.stamina = game.player1.sprinting ? 
                Math.max(0, game.player1.stamina - game.staminaDrainRate) :
                Math.min(game.maxStamina, game.player1.stamina + game.staminaRegenRate);
            
            game.player2.stamina = game.player2.sprinting ? 
                Math.max(0, game.player2.stamina - game.staminaDrainRate) :
                Math.min(game.maxStamina, game.player2.stamina + game.staminaRegenRate);

            // 공 물리
            game.ball.x += game.ball.vx;
            game.ball.y += game.ball.vy;
            game.ball.vx *= 0.985;
            game.ball.vy *= 0.985;

            // 공 벽 충돌
            if (game.ball.x < game.ballRadius || game.ball.x > game.fieldWidth - game.ballRadius) {
                game.ball.vx *= -0.8;
                game.ball.x = game.ball.x < game.ballRadius ? game.ballRadius : game.fieldWidth - game.ballRadius;
            }
            if (game.ball.y < game.ballRadius || game.ball.y > game.fieldHeight - game.ballRadius) {
                game.ball.vy *= -0.8;
                game.ball.y = game.ball.y < game.ballRadius ? game.ballRadius : game.fieldHeight - game.ballRadius;
            }

            // 플레이어 1 충돌
            if (checkCircleCollision(game.ball, game.player1, game.ballRadius, game.playerRadius)) {
                const angle = Math.atan2(game.ball.y - game.player1.y, game.ball.x - game.player1.x);
                const baseSpeed = 10 * SCALE;
                const speed = game.player1.sprinting ? baseSpeed * 1.8 : baseSpeed;
                game.ball.vx = Math.cos(angle) * speed;
                game.ball.vy = Math.sin(angle) * speed;
                
                if (game.player1.sprinting) {
                    game.player1.stamina = Math.max(0, game.player1.stamina - game.staminaKickCost);
                }
                
                const dist = Math.sqrt(Math.pow(game.ball.x - game.player1.x, 2) + Math.pow(game.ball.y - game.player1.y, 2));
                const overlap = game.ballRadius + game.playerRadius - dist;
                game.ball.x += Math.cos(angle) * overlap;
                game.ball.y += Math.sin(angle) * overlap;
            }

            // 플레이어 2 충돌
            if (checkCircleCollision(game.ball, game.player2, game.ballRadius, game.playerRadius)) {
                const angle = Math.atan2(game.ball.y - game.player2.y, game.ball.x - game.player2.x);
                const baseSpeed = 10 * SCALE;
                const speed = game.player2.sprinting ? baseSpeed * 1.8 : baseSpeed;
                game.ball.vx = Math.cos(angle) * speed;
                game.ball.vy = Math.sin(angle) * speed;
                
                if (game.player2.sprinting) {
                    game.player2.stamina = Math.max(0, game.player2.stamina - game.staminaKickCost);
                }
                
                const dist = Math.sqrt(Math.pow(game.ball.x - game.player2.x, 2) + Math.pow(game.ball.y - game.player2.y, 2));
                const overlap = game.ballRadius + game.playerRadius - dist;
                game.ball.x += Math.cos(angle) * overlap;
                game.ball.y += Math.sin(angle) * overlap;
            }

            // 골 감지
            const goalTop = (game.fieldHeight - game.goalWidth) / 2;
            const goalBottom = goalTop + game.goalWidth;

            if (game.ball.x <= game.goalDepth && game.ball.y >= goalTop && game.ball.y <= goalBottom) {
                score.right++;
                document.getElementById('scoreRight').textContent = score.right;
                resetBall();
                resetPlayers();
                gameStarted = false;
                document.getElementById('startHint').style.display = 'block';
            }

            if (game.ball.x >= game.fieldWidth - game.goalDepth && game.ball.y >= goalTop && game.ball.y <= goalBottom) {
                score.left++;
                document.getElementById('scoreLeft').textContent = score.left;
                resetBall();
                resetPlayers();
                gameStarted = false;
                document.getElementById('startHint').style.display = 'block';
            }

            // 그리기
            draw(goalTop, goalBottom);
        }

        function draw(goalTop, goalBottom) {
            ctx.clearRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);

            // 필드 배경
            ctx.fillStyle = '#2d8a3e';
            ctx.fillRect(0, 0, game.fieldWidth, game.fieldHeight);

            // 필드 줄무늬
            ctx.fillStyle = '#248a34';
            for (let i = 1; i < 10; i += 2) {
                ctx.fillRect(i * (game.fieldWidth / 10), 0, game.fieldWidth / 10, game.fieldHeight);
            }

            // 필드 테두리
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 5 * SCALE;
            ctx.strokeRect(3, 3, game.fieldWidth - 6, game.fieldHeight - 6);

            // 센터라인
            ctx.lineWidth = 3 * SCALE;
            ctx.beginPath();
            ctx.moveTo(game.fieldWidth / 2, 0);
            ctx.lineTo(game.fieldWidth / 2, game.fieldHeight);
            ctx.stroke();

            // 센터 원
            ctx.beginPath();
            ctx.arc(game.fieldWidth / 2, game.fieldHeight / 2, 100 * SCALE, 0, Math.PI * 2);
            ctx.stroke();

            // 센터 점
            ctx.fillStyle = '#fff';
            ctx.beginPath();
            ctx.arc(game.fieldWidth / 2, game.fieldHeight / 2, 5 * SCALE, 0, Math.PI * 2);
            ctx.fill();

            // 골
            ctx.fillStyle = '#1565c0';
            ctx.fillRect(0, goalTop, game.goalDepth, game.goalWidth);
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 4 * SCALE;
            ctx.strokeRect(0, goalTop, game.goalDepth, game.goalWidth);

            ctx.fillStyle = '#c62828';
            ctx.fillRect(game.fieldWidth - game.goalDepth, goalTop, game.goalDepth, game.goalWidth);
            ctx.strokeRect(game.fieldWidth - game.goalDepth, goalTop, game.goalDepth, game.goalWidth);

            // 골 에어리어
            ctx.lineWidth = 3 * SCALE;
            ctx.strokeRect(0, goalTop - 60 * SCALE, 120 * SCALE, game.goalWidth + 120 * SCALE);
            ctx.strokeRect(game.fieldWidth - 120 * SCALE, goalTop - 60 * SCALE, 120 * SCALE, game.goalWidth + 120 * SCALE);

            // 플레이어 1 (파랑)
            if (game.player1.sprinting) {
                ctx.shadowColor = '#2196F3';
                ctx.shadowBlur = 30 * SCALE;
            }
            ctx.fillStyle = '#2196F3';
            ctx.beginPath();
            ctx.arc(game.player1.x, game.player1.y, game.playerRadius, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;
            ctx.strokeStyle = '#0d47a1';
            ctx.lineWidth = 3 * SCALE;
            ctx.stroke();
            
            ctx.fillStyle = '#fff';
            ctx.beginPath();
            ctx.arc(game.player1.x, game.player1.y - 8 * SCALE, 3 * SCALE, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.font = `bold ${16 * SCALE}px Arial`;
            ctx.textAlign = 'center';
            ctx.fillText('P1', game.player1.x, game.player1.y + 6 * SCALE);

            // 플레이어 2 (빨강)
            if (game.player2.sprinting) {
                ctx.shadowColor = '#F44336';
                ctx.shadowBlur = 30 * SCALE;
            }
            ctx.fillStyle = '#F44336';
            ctx.beginPath();
            ctx.arc(game.player2.x, game.player2.y, game.playerRadius, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;
            ctx.strokeStyle = '#b71c1c';
            ctx.lineWidth = 3 * SCALE;
            ctx.stroke();
            
            ctx.fillStyle = '#fff';
            ctx.beginPath();
            ctx.arc(game.player2.x, game.player2.y - 8 * SCALE, 3 * SCALE, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.fillText('P2', game.player2.x, game.player2.y + 6 * SCALE);

            // 공
            ctx.fillStyle = '#fff';
            ctx.beginPath();
            ctx.arc(game.ball.x, game.ball.y, game.ballRadius, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.strokeStyle = '#000';
            ctx.lineWidth = 2 * SCALE;
            ctx.stroke();
            ctx.beginPath();
            ctx.moveTo(game.ball.x - 6 * SCALE, game.ball.y);
            ctx.lineTo(game.ball.x + 6 * SCALE, game.ball.y);
            ctx.moveTo(game.ball.x, game.ball.y - 6 * SCALE);
            ctx.lineTo(game.ball.x, game.ball.y + 6 * SCALE);
            ctx.stroke();

            // 스태미나 바
            const barWidth = 150 * SCALE;
            const barHeight = 20 * SCALE;
            
            // 플레이어 1 스태미나
            ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
            ctx.fillRect(game.player1.x - barWidth / 2, game.player1.y - 50 * SCALE, barWidth, barHeight);
            ctx.fillStyle = game.player1.stamina > 30 ? '#4CAF50' : '#F44336';
            ctx.fillRect(game.player1.x - barWidth / 2, game.player1.y - 50 * SCALE, (game.player1.stamina / game.maxStamina) * barWidth, barHeight);
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 2 * SCALE;
            ctx.strokeRect(game.player1.x - barWidth / 2, game.player1.y - 50 * SCALE, barWidth, barHeight);

            // 플레이어 2 스태미나
            ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
            ctx.fillRect(game.player2.x - barWidth / 2, game.player2.y - 50 * SCALE, barWidth, barHeight);
            ctx.fillStyle = game.player2.stamina > 30 ? '#4CAF50' : '#F44336';
            ctx.fillRect(game.player2.x - barWidth / 2, game.player2.y - 50 * SCALE, (game.player2.stamina / game.maxStamina) * barWidth, barHeight);
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 2 * SCALE;
            ctx.strokeRect(game.player2.x - barWidth / 2, game.player2.y - 50 * SCALE, barWidth, barHeight);
        }

        function resetGame() {
            score = { left: 0, right: 0 };
            document.getElementById('scoreLeft').textContent = '0';
            document.getElementById('scoreRight').textContent = '0';
            gameStarted = false;
            document.getElementById('startHint').style.display = 'block';
            game.player1 = { x: 400 * SCALE, y: 600 * SCALE, stamina: 100, sprinting: false };
            game.player2 = { x: 1200 * SCALE, y: 600 * SCALE, stamina: 100, sprinting: false };
            game.ball = { x: 800 * SCALE, y: 600 * SCALE, vx: 0, vy: 0 };
        }

        // 게임 루프 시작
        setInterval(gameLoop, 1000 / 60);
        
        // 초기 화면 그리기
        draw((game.fieldHeight - game.goalWidth) / 2, (game.fieldHeight - game.goalWidth) / 2 + game.goalWidth);
    </script>
